document.addEventListener('DOMContentLoaded', function () {
    document.querySelectorAll('.pc-content p').forEach(function (zeile) {
        const text = zeile.textContent.trim();
        const kapitelnummer = text.match(/^(\d+(?:\.\d+)*)\s*[–-]/);

        if (!kapitelnummer) {
            return;
        }

        const ebene = kapitelnummer[1].split('.').length;

        if (ebene === 1) {
            zeile.style.setProperty('margin-left', '0', 'important');
            zeile.style.setProperty('font-size', '1.35rem', 'important'); 
            zeile.style.setProperty('margin-top', '35px', 'important');
            zeile.style.setProperty('font-weight', '600', 'important');
            zeile.style.setProperty('line-height', '1.3', 'important');
        } else if (ebene === 2) {
            zeile.style.setProperty('margin-left', '24px', 'important');
            zeile.style.setProperty('font-size', '1.05rem', 'important');

            zeile.style.setProperty('margin-top', '0', 'important');
            zeile.style.setProperty('font-weight', '400', 'important');
        }
    });
